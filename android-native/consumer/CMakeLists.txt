cmake_minimum_required(VERSION 3.20)
project(MadObsAndroidConsumer LANGUAGES C CXX)
if(NOT ANDROID OR NOT ANDROID_ABI STREQUAL "arm64-v8a")
  message(FATAL_ERROR "Android ARM64 only")
endif()
# Only this SDK config lookup bypasses cross-root rewriting. Dependencies such as
# ZLIB keep the NDK target search policy and explicit sysroot inputs.
if(NOT DEFINED MadObs_DIR OR NOT EXISTS "${MadObs_DIR}/MadObsConfig.cmake")
  message(FATAL_ERROR "MADOBS_EXACT_CONFIG_REQUIRED")
endif()
file(REAL_PATH "${MadObs_DIR}/MadObsConfig.cmake" expected_madobs_config)
find_package(MadObs 0.1.0 EXACT CONFIG REQUIRED
  PATHS "${MadObs_DIR}" NO_DEFAULT_PATH NO_CMAKE_FIND_ROOT_PATH)
file(REAL_PATH "${MadObs_CONFIG}" actual_madobs_config)
if(NOT actual_madobs_config STREQUAL expected_madobs_config)
  message(FATAL_ERROR "MADOBS_EXACT_CONFIG_MISMATCH")
endif()
add_executable(madobs_c_smoke smoke.c)
set_target_properties(madobs_c_smoke PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES LINKER_LANGUAGE CXX)
target_link_libraries(madobs_c_smoke PRIVATE MadObs::runtime)
target_link_options(madobs_c_smoke PRIVATE -Wl,-z,max-page-size=16384 -Wl,-z,common-page-size=16384 -Wl,--no-undefined)
