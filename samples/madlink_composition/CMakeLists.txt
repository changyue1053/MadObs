cmake_minimum_required(VERSION 3.20)
project(MadObsMadLinkComposition LANGUAGES C CXX)
find_package(MadObs 0.1.0 EXACT REQUIRED)
if(NOT TARGET MadObs::BoringSSL)
  message(FATAL_ERROR "This smoke requires the verified-provider MadObs installation")
endif()
foreach(input MADLINK_SOURCE MADLINK_ARCHIVE MADLINK_CORE_ARCHIVE BORINGSSL_SSL_ARCHIVE)
  if(NOT EXISTS "${${input}}")
    message(FATAL_ERROR "Provide existing ${input}")
  endif()
endforeach()
add_executable(composition main.cc)
target_compile_features(composition PRIVATE cxx_std_17)
target_compile_options(composition PRIVATE -Wall -Wextra -Werror)
target_include_directories(composition PRIVATE "${MADLINK_SOURCE}/include" "${MADLINK_SOURCE}/imlink/include" "${MADLINK_SOURCE}/adapters/obs/include")
# A single crypto provider, from MadObs::runtime. The supplied ssl archive must
# have the same upstream revision. No whole-archive or symbol-isolation claim.
target_link_libraries(composition PRIVATE "${MADLINK_ARCHIVE}" "${MADLINK_CORE_ARCHIVE}" "${BORINGSSL_SSL_ARCHIVE}" MadObs::runtime)
enable_testing()
add_test(NAME composition COMMAND composition)
