cmake_minimum_required(VERSION 3.20)
project(MadObsInstalledHolder LANGUAGES C CXX)
if(NOT IS_ABSOLUTE "${SDK_PREFIX}")
  message(FATAL_ERROR "An absolute relocated SDK prefix is required")
endif()
find_package(MadObs 0.1.0 EXACT CONFIG REQUIRED PATHS "${SDK_PREFIX}" NO_DEFAULT_PATH)
add_executable(holder main.c)
set_target_properties(holder PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
target_compile_options(holder PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_link_libraries(holder PRIVATE MadObs::runtime)
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/runtime-path" CONTENT "$<TARGET_FILE:MadObs::runtime>")
