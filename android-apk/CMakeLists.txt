cmake_minimum_required(VERSION 3.20)
project(MadObsNativePackage LANGUAGES C CXX)
if(NOT ANDROID OR NOT ANDROID_ABI STREQUAL "arm64-v8a")
  message(FATAL_ERROR "Android ARM64 only")
endif()
if(NOT DEFINED MadObs_DIR OR NOT EXISTS "${MadObs_DIR}/MadObsConfig.cmake")
  message(FATAL_ERROR "Exact installed MadObs required")
endif()
file(REAL_PATH "${MadObs_DIR}/MadObsConfig.cmake" expected)
find_package(MadObs 0.1.0 EXACT CONFIG REQUIRED PATHS "${MadObs_DIR}" NO_DEFAULT_PATH NO_CMAKE_FIND_ROOT_PATH)
file(REAL_PATH "${MadObs_CONFIG}" actual)
if(NOT actual STREQUAL expected)
  message(FATAL_ERROR "Exact installed MadObs mismatch")
endif()
add_library(madobs_preview SHARED activity.c)
set_target_properties(madobs_preview PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES LINKER_LANGUAGE CXX)
target_link_libraries(madobs_preview PRIVATE MadObs::runtime android)
target_link_options(madobs_preview PRIVATE -Wl,--no-undefined -Wl,-z,max-page-size=16384 -Wl,-z,common-page-size=16384)
