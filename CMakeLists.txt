cmake_minimum_required(VERSION 3.20)
project(MadObs VERSION 0.1.0 LANGUAGES C CXX)
include(CTest)
add_compile_options(-UNDEBUG)
if(IMOBS_SAN STREQUAL "asan")
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
elseif(IMOBS_SAN STREQUAL "tsan")
  add_compile_options(-fsanitize=thread -fno-omit-frame-pointer)
  add_link_options(-fsanitize=thread)
endif()
add_library(madobs_api INTERFACE)
add_library(MadObs::api ALIAS madobs_api)
set_target_properties(madobs_api PROPERTIES EXPORT_NAME api)
target_include_directories(madobs_api INTERFACE $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
option(MADOBS_BUILD_SHARED "Build shared runtime alongside static archive" ON)
option(MADOBS_BUILD_RUNTIME "Build preserved native runtime" ON)
if(MADOBS_BUILD_RUNTIME)
  add_subdirectory(imobs)
  add_library(MadObs::runtime ALIAS imobs_core)
  set_target_properties(imobs_core PROPERTIES EXPORT_NAME runtime OUTPUT_NAME madobs_runtime)
  install(TARGETS imobs_core EXPORT MadObsRuntimeTargets ARCHIVE DESTINATION lib)
  install(EXPORT MadObsRuntimeTargets NAMESPACE MadObs:: DESTINATION lib/cmake/MadObs)
  install(FILES imobs/include/imobs/imobs_c.h DESTINATION include/imobs)
  install(FILES include/madobs/runtime.h include/madobs/private.h include/madobs/private_runtime.hpp include/madobs/private_wal.hpp DESTINATION include/madobs)
endif()
if(BUILD_TESTING)
  foreach(mode strict noexceptions)
    add_executable(test_madobs_privacy_${mode} tests/contract/test_privacy_cpp.cc)
    target_compile_options(test_madobs_privacy_${mode} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    set_target_properties(test_madobs_privacy_${mode} PROPERTIES CXX_STANDARD 11 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
    target_link_libraries(test_madobs_privacy_${mode} PRIVATE MadObs::api)
    if(mode STREQUAL "noexceptions")
      target_compile_options(test_madobs_privacy_${mode} PRIVATE -fno-exceptions)
    endif()
    add_test(NAME madobs_privacy_${mode} COMMAND test_madobs_privacy_${mode})
  endforeach()
  add_executable(test_madobs_privacy tests/contract/test_privacy.c)
  target_link_libraries(test_madobs_privacy PRIVATE MadObs::api)
  add_test(NAME madobs_privacy COMMAND test_madobs_privacy)
  add_executable(test_madobs_api tests/contract/test_api.c)
  target_link_libraries(test_madobs_api PRIVATE MadObs::api)
  add_test(NAME madobs_api_c COMMAND test_madobs_api)
  add_executable(test_madobs_api_cpp tests/contract/test_api_cpp.cc)
  target_link_libraries(test_madobs_api_cpp PRIVATE MadObs::api)
  add_test(NAME madobs_api_cpp COMMAND test_madobs_api_cpp)
  add_executable(test_madobs_api_noexceptions tests/contract/test_api_noexceptions.cc)
  target_compile_options(test_madobs_api_noexceptions PRIVATE -fno-exceptions)
  target_link_libraries(test_madobs_api_noexceptions PRIVATE MadObs::api)
  add_test(NAME madobs_api_noexceptions COMMAND test_madobs_api_noexceptions)
  if(MADOBS_BUILD_RUNTIME)
    add_executable(test_madobs_regressions tests/contract/test_regressions.cc)
    target_compile_features(test_madobs_regressions PRIVATE cxx_std_20)
    target_include_directories(test_madobs_regressions PRIVATE imobs/test)
    target_link_libraries(test_madobs_regressions PRIVATE MadObs::runtime)
    add_test(NAME madobs_regressions COMMAND test_madobs_regressions)
    if(UNIX AND NOT APPLE)
      add_executable(test_madobs_codec_faults tests/contract/test_codec_faults.cc)
      target_compile_features(test_madobs_codec_faults PRIVATE cxx_std_20)
      target_link_libraries(test_madobs_codec_faults PRIVATE MadObs::runtime)
      target_link_options(test_madobs_codec_faults PRIVATE -Wl,--wrap=compress2 -Wl,--wrap=uncompress -Wl,--wrap=write -Wl,--wrap=fsync)
      add_test(NAME madobs_codec_faults COMMAND test_madobs_codec_faults)
    endif()
    add_executable(test_madobs_private_c tests/contract/test_private_c.c)
    target_link_libraries(test_madobs_private_c PRIVATE MadObs::runtime)
    add_test(NAME madobs_private_c COMMAND test_madobs_private_c)
    add_executable(test_madobs_private_runtime tests/contract/test_private_runtime.cc)
    target_compile_features(test_madobs_private_runtime PRIVATE cxx_std_20)
    target_include_directories(test_madobs_private_runtime PRIVATE imobs/test)
    target_compile_options(test_madobs_private_runtime PRIVATE -Wall -Wextra -Wpedantic -Werror)
    target_link_libraries(test_madobs_private_runtime PRIVATE MadObs::runtime)
    add_test(NAME madobs_private_runtime COMMAND test_madobs_private_runtime)
    add_executable(test_madobs_bridge tests/contract/test_runtime_bridge.cc)
    target_compile_features(test_madobs_bridge PRIVATE cxx_std_20)
    target_include_directories(test_madobs_bridge PRIVATE imobs/test)
    target_link_libraries(test_madobs_bridge PRIVATE MadObs::runtime)
    add_test(NAME madobs_bridge COMMAND test_madobs_bridge)
  endif()
endif()
install(FILES include/madobs/observer.h include/madobs/privacy.h DESTINATION include/madobs)
install(TARGETS madobs_api EXPORT MadObsApiTargets)
install(EXPORT MadObsApiTargets NAMESPACE MadObs:: DESTINATION lib/cmake/MadObs)

install(FILES cmake/MadObsConfig.cmake DESTINATION lib/cmake/MadObs)

# Experimental SDK: only equal requested versions are compatible; unversioned
# discovery remains supported for existing consumers, with no stable ABI promise.
include(CMakePackageConfigHelpers)
if(MADOBS_BUILD_RUNTIME)
  write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/MadObsConfigVersion.cmake"
    VERSION ${PROJECT_VERSION} COMPATIBILITY ExactVersion)
else()
  write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/MadObsConfigVersion.cmake"
    VERSION ${PROJECT_VERSION} COMPATIBILITY ExactVersion ARCH_INDEPENDENT)
endif()
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/MadObsConfigVersion.cmake" DESTINATION lib/cmake/MadObs)
